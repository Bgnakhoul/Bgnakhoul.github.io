Bluetooth UART · bring-up and receive

; 9600 baud, 8N1, continuous receive
MOVLW B'00100100'    ; TXEN, BRGH = 1
MOVWF TXSTA
MOVLW D'25'          ; SPBRG
MOVWF SPBRG
MOVLW B'10010000'    ; SPEN, CREN
MOVWF RCSTA

LOOP1
  BTFSC RCSTA,OERR     ; overrun?
  GOTO  FIX_OERR       ; pulse CREN
WAIT_RX
  BTFSS PIR1,RCIF      ; byte received?
  GOTO  WAIT_RX
  MOVF   RCREG,W       ; target room

Line following · computed jump

READ_IR
  BTFSS PIR1,TXIF      ; TXREG free?
  GOTO  $-1
  MOVF   ROOM,W
  MOVWF  TXREG         ; status byte to the PC
  MOVF   PORTB,W       ; 3 IR sensors
  ANDLW  H'07'
  ADDWF  PCL,F         ; computed jump
  GOTO   READ_IR       ; 000
  GOTO   RIGHT         ; 001
  GOTO   STRAIGHT      ; 010
  GOTO   RIGHT         ; 011
  GOTO   LEFT          ; 100
  GOTO   DECREMENT     ; 101 room marker
  GOTO   LEFT          ; 110
  GOTO   READ_IR       ; 111
